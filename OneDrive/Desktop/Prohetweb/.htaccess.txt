# Enable mod_rewrite
RewriteEngine On

# Set the base path for the application (if necessary, replace '/Prohetweb/' with the correct path)
RewriteBase /Prohetweb/

# Redirect to index.php for all routes
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d
RewriteRule ^(.*)$ index.php?url=$1 [QSA,L]
